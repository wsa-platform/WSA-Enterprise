<?php

namespace App\Services\Agriculture\Research\Projection;

use App\Services\Agriculture\Research\Capability\CapabilityState;

/**
 * Per-facet query constraint derived from the facet's Cap `query_constrainable` cell (ADR-023 §8.33).
 *
 * Read-only mapping: assigns no Cap state and carries no query text. A NOT_CONSTRAINABLE
 * facet the Builder still places as plain lexical text is APPROXIMATED with a reason in the
 * facet accounting, never MAPPED.
 */
enum ProjectionFacetConstraintState: string
{
    case CONSTRAINABLE = 'CONSTRAINABLE';
    case NOT_CONSTRAINABLE = 'NOT_CONSTRAINABLE';
    case UNKNOWN = 'UNKNOWN';

    /**
     * @param  CapabilityState|null  $state  null when no query_constrainable record exists (MISSING)
     */
    public static function fromCapabilityState(?CapabilityState $state): self
    {
        return match ($state) {
            CapabilityState::VERIFIED => self::CONSTRAINABLE,
            CapabilityState::UNAVAILABLE, CapabilityState::NOT_APPLICABLE => self::NOT_CONSTRAINABLE,
            CapabilityState::UNVERIFIED, CapabilityState::PARTIAL, null => self::UNKNOWN,
        };
    }
}

<?php

namespace App\Services\Agriculture\Research\Projection;

use App\Services\Agriculture\Research\Capability\CapabilityState;

/**
 * One facet entry of ProjectionQueryConstraints (ADR-023 §8.33).
 *
 * identity_critical and constraint are derived, never supplied by the caller.
 */
final readonly class ProjectionFacetConstraint
{
    public const CAPABILITY_STATE_MISSING = 'MISSING';

    private function __construct(
        public string $facetCode,
        public bool $identityCritical,
        public string $capabilityState,
        public ProjectionFacetConstraintState $constraint,
    ) {}

    /**
     * @param  CapabilityState|null  $capabilityState  null when no query_constrainable record exists (MISSING)
     */
    public static function of(string $facetCode, ?CapabilityState $capabilityState): self
    {
        $code = ProjectionIdentityCriticalFacets::normalize($facetCode);

        return new self(
            $code,
            ProjectionIdentityCriticalFacets::isIdentityCritical($code),
            $capabilityState?->value ?? self::CAPABILITY_STATE_MISSING,
            ProjectionFacetConstraintState::fromCapabilityState($capabilityState),
        );
    }

    public function isCapabilityMissing(): bool
    {
        return $this->capabilityState === self::CAPABILITY_STATE_MISSING;
    }

    /**
     * @return array{facet_code: string, identity_critical: bool, capability_state: string, constraint: string}
     */
    public function toArray(): array
    {
        return [
            'facet_code' => $this->facetCode,
            'identity_critical' => $this->identityCritical,
            'capability_state' => $this->capabilityState,
            'constraint' => $this->constraint->value,
        ];
    }
}

<?php

namespace App\Services\Agriculture\Research\Projection;

use App\Services\Agriculture\Research\Capability\CapabilityDecisionIdentity;
use App\Services\Agriculture\Research\Identity\AdrMembershipId;
use App\Services\Agriculture\Research\Identity\CanonicalSourceIdentityId;
use App\Services\Agriculture\Research\Path\PathDecisionIdentity;
use App\Services\Agriculture\Research\Path\PathId;

/**
 * Per-facet query constraint metadata for one seat and one CSQ (ADR-023 §8.33, B-1).
 *
 * Consumed by ScientificSearchQueryBuilder, which remains the sole lexical construction and
 * expansion authority (D-04, D-10.9.2). Carries no query text, terms, expansions or fidelity:
 * source_query, expanded_terms and fidelity_class stay in ProjectionEnvelope.
 * In-memory only; not wired into runtime.
 */
final readonly class ProjectionQueryConstraints
{
    public const SCHEMA_VERSION = 'projection_query_constraints.v1';

    /**
     * @param  list<ProjectionFacetConstraint>  $facetConstraints
     * @param  list<ProjectionWarning>  $warnings
     */
    private function __construct(
        public CanonicalQueryId $canonicalQueryId,
        public AdrMembershipId $adrId,
        public ?CanonicalSourceIdentityId $canonicalIdentityId,
        public string $capabilityDecisionId,
        public string $pathDecisionId,
        public PathId $pathId,
        public array $facetConstraints,
        public ProjectionRetrievalMethod $retrievalMethod,
        public array $warnings,
    ) {}

    /**
     * @param  list<ProjectionFacetConstraint>  $facetConstraints
     * @param  list<ProjectionWarning>  $warnings
     */
    public static function create(
        CanonicalQueryId $canonicalQueryId,
        CapabilityDecisionIdentity $capabilityDecision,
        PathDecisionIdentity $pathDecision,
        array $facetConstraints,
        ?ProjectionRetrievalMethod $retrievalMethod = null,
        array $warnings = [],
    ): self {
        ProjectionDomainContract::assertSubjectAlignment($capabilityDecision, $pathDecision);

        $byCode = [];
        foreach ($facetConstraints as $constraint) {
            if (! $constraint instanceof ProjectionFacetConstraint) {
                throw new ProjectionInvariantViolation(
                    'facet_constraints must be ProjectionFacetConstraint instances.'
                );
            }
            if (isset($byCode[$constraint->facetCode])) {
                throw new ProjectionInvariantViolation(
                    "Duplicate facet_code [{$constraint->facetCode}] in facet_constraints."
                );
            }
            $byCode[$constraint->facetCode] = $constraint;
        }
        ksort($byCode, SORT_STRING);

        foreach ($warnings as $warning) {
            if (! $warning instanceof ProjectionWarning) {
                throw new ProjectionInvariantViolation('warnings must be ProjectionWarning instances.');
            }
        }

        return new self(
            $canonicalQueryId,
            $pathDecision->adrId,
            $pathDecision->canonicalIdentityId,
            $capabilityDecision->decisionId,
            $pathDecision->decisionId->value,
            $pathDecision->pathId,
            array_values($byCode),
            $retrievalMethod ?? ProjectionRetrievalMethod::unknown(),
            array_values($warnings),
        );
    }

    /**
     * @return array<string, mixed>
     */
    public function toArray(): array
    {
        return [
            'schema_version' => self::SCHEMA_VERSION,
            'canonical_query_id' => $this->canonicalQueryId->value,
            'adr_id' => $this->adrId->value,
            'canonical_identity_id' => $this->canonicalIdentityId?->value,
            'capability_decision_id' => $this->capabilityDecisionId,
            'path_decision_id' => $this->pathDecisionId,
            'path_id' => $this->pathId->value,
            'facet_constraints' => array_map(
                static fn (ProjectionFacetConstraint $c) => $c->toArray(),
                $this->facetConstraints,
            ),
            'retrieval_method' => $this->retrievalMethod->value,
            'warnings' => array_map(
                static fn (ProjectionWarning $w) => $w->toArray(),
                $this->warnings,
            ),
        ];
    }
}

<?php

namespace App\Services\Agriculture\Research\Projection;

use App\Services\Agriculture\Research\Capability\CapabilityDecisionIdentity;
use App\Services\Agriculture\Research\Capability\CapabilityRecord;
use App\Services\Agriculture\Research\Capability\CapabilityState;
use App\Services\Agriculture\Research\Path\PathDecisionIdentity;

/**
 * Sole producer of ProjectionQueryConstraints (ADR-023 §8.33, B-1).
 *
 * Facet codes are explicit caller input: no approved contract yet defines their extraction
 * from CSQ. Reads Cap records only; never writes Cap, CapVer, or any repository.
 */
final class ProjectionQueryConstraintsFactory
{
    public const WARNING_QUERY_CONSTRAINT_UNKNOWN = 'QUERY_CONSTRAINT_UNKNOWN';

    /**
     * @param  list<string>  $facetCodes
     * @param  list<CapabilityRecord>  $capabilityRecords
     * @param  list<ProjectionWarning>  $warnings
     */
    public function derive(
        CanonicalQueryId $canonicalQueryId,
        CapabilityDecisionIdentity $capabilityDecision,
        PathDecisionIdentity $pathDecision,
        array $facetCodes,
        array $capabilityRecords,
        ?ProjectionRetrievalMethod $retrievalMethod = null,
        array $warnings = [],
    ): ProjectionQueryConstraints {
        ProjectionDomainContract::assertSubjectAlignment($capabilityDecision, $pathDecision);

        $codes = [];
        foreach ($facetCodes as $facetCode) {
            if (! is_string($facetCode)) {
                throw new ProjectionInvariantViolation('Facet codes must be strings.');
            }
            $code = ProjectionIdentityCriticalFacets::normalize($facetCode);
            if (isset($codes[$code])) {
                throw new ProjectionInvariantViolation("Duplicate facet_code [{$code}] in facet codes.");
            }
            $codes[$code] = true;
        }
        $codes = array_keys($codes);
        sort($codes, SORT_STRING);

        $statesByCode = $this->queryConstrainableStates($capabilityDecision, $capabilityRecords);

        $constraints = [];
        $generated = [];
        foreach ($codes as $code) {
            $constraint = ProjectionFacetConstraint::of($code, $statesByCode[$code] ?? null);
            $constraints[] = $constraint;
            if ($constraint->constraint === ProjectionFacetConstraintState::UNKNOWN) {
                $generated[] = ProjectionWarning::of(
                    self::WARNING_QUERY_CONSTRAINT_UNKNOWN,
                    "Facet [{$code}] query constraint is UNKNOWN (capability_state {$constraint->capabilityState}).",
                );
            }
        }

        return ProjectionQueryConstraints::create(
            $canonicalQueryId,
            $capabilityDecision,
            $pathDecision,
            $constraints,
            $retrievalMethod,
            [...array_values($warnings), ...$generated],
        );
    }

    /**
     * @param  list<CapabilityRecord>  $capabilityRecords
     * @return array<string, CapabilityState>
     */
    private function queryConstrainableStates(
        CapabilityDecisionIdentity $capabilityDecision,
        array $capabilityRecords,
    ): array {
        $subject = $capabilityDecision->subject;
        $states = [];
        foreach ($capabilityRecords as $record) {
            if (! $record instanceof CapabilityRecord) {
                throw new ProjectionInvariantViolation('Capability records must be CapabilityRecord instances.');
            }
            if (! $record->dimension->isQueryConstrainableFacet()) {
                continue;
            }
            if (! $record->subject->adrId->equals($subject->adrId)
                || $record->subject->canonicalIdentityId?->value !== $subject->canonicalIdentityId?->value
            ) {
                throw new ProjectionInvariantViolation(
                    'Capability record subject must match the capability decision subject.'
                );
            }
            $code = ProjectionIdentityCriticalFacets::normalize($record->dimension->code);
            if (isset($states[$code])) {
                throw new ProjectionInvariantViolation(
                    "Ambiguous query_constrainable capability records for facet_code [{$code}]."
                );
            }
            $states[$code] = $record->state;
        }

        return $states;
    }
}

## 4. الملاحظات المصنّفة

### مانع اعتماد

**M-1 — ‏`VERIFIED` يُحوَّل إلى `CONSTRAINABLE` دون النظر إلى حداثة الدليل.**
- **الموضع:** المصنع سطر 105 (`$states[$code] = $record->state;`)، والحالة سطر 26.
- **التعارض:** ‏§8.23.3 تقول: "VERIFIED requires CURRENT evidence. STALE or UNKNOWN freshness never yields VERIFIED" (ADR 4274). ‏D-10.5.2 تقول: "STALE evidence must not auto-promote to VERIFIED" (ADR 1415). تعليق `CapabilityRecord.php` في الأسطر 57–60 يقول: "consumers must not treat as current verified automation".
- **أثر الاختبارات:** الدالة المساعدة في الاختبار (572–580) تنشئ سجلات VERIFIED بحداثة `UNKNOWN` الافتراضية، وتتوقع `CONSTRAINABLE` (السطر 83 و290–293). أي أن الاختبارات تثبّت السلوك المخالف.
- **القرار المطلوب** (لا أخترعه):
  - **(أ)** ‏VERIFIED بحداثة غير `CURRENT` ← `UNKNOWN` مع تحذير؛
  - **(ب)** ‏`CONSTRAINABLE` مع تحذير حداثة؛
  - **(ج)** تأكيد أن الحداثة شأن Capability وحده.

  في حالتي (أ) و(ب) يجب أن تستخدم أدوات إعداد الاختبارات حداثة `CURRENT`.

### مخاطر تحتاج قرارًا

**R-1 — المصنع يقرأ سجلات Capability خامًا ولا يرتبط بلقطة القرار نفسه.**
- **الموضع:** المصنع 50 و75–108.
- **الانحراف عن المقيِّم:** ‏`CapabilityRequirementEvaluator` (الأسطر 131–152) يتجاهل سجلات المواضيع الأخرى، ويحسم السجلات المكررة بقاعدة "الأكثر تقييدًا يغلب" (seat override مقابل الخلية المشتركة). المصنع بدلًا من ذلك يرفض الحالتين (السطر 92–98 والسطر 100–104).
- **الأثر:** مدخلات يقبلها Capability تجعل المصنع يرمي استثناءً. وقد لا تطابق القيود القرار المذكور في `capability_decision_id`، مع أن الحزمة تصفه بأنه القرار "the constraints were derived from" (الحزمة سطر 95).
- **الأساس الحالي:** خيار التنفيذ 3 في ADR (سطر 5669) سجّل هذا الرفض، وتصميم الحزمة (سطر 156) نص على قائمة `CapabilityRecord`.
- **القرار المطلوب:** مصدر الحالة. الخيارات:
  - `perCellOutcomes` من قرار Capability؛
  - أو حسم Capability نفسه للتكرار؛
  - أو إبقاء الرفض.

**R-2 — مطابقة رموز أبعاد Capability دون حساسية لحالة الأحرف.**
- **الموضع:** المصنع سطر 99 يحوّل `dimension->code` إلى أحرف صغيرة.
- **التعارض:** ‏Capability تعامل الرمز كمعرّف معتم حساس لحالة الأحرف؛ `normalizeCode` يقص المسافات فقط، و`key()` يجمع العائلة والرمز.
- **الأثر:** خليتان مختلفتان لدى Capability (`Species` و`species`) تُرفضان عند Projection كـ "Ambiguous". الخطورة منخفضة، لأن الأثر رفض لا تحويل صامت، لكنها دلالة غير موثقة في §8.33.2.

**R-3 — صياغات في §8.33 تبدو قرارات مالك وهي في الحقيقة تقديرات المنفّذ.**
- **ADR سطر 5636:** عبارة "For query construction it replaces the phrase 'Projection owns query'". تعليمات 6:22 لم تذكر T2، فهذا استنتاج مني غير موسوم بـ [I].
- **ADR الأسطر 5673–5688:** بنود GA-B1-01 صاغها المنفّذ دون الإشارة إلى ذلك، بخلاف §8.30.3 (سطر 5293).
- **ADR الأسطر 5744–5748:** حالات "PARTIALLY MET" و"MET FOR THE B-1 SCOPE" تقديرات المنفّذ.
- **القرار المطلوب:** تأكيدها، أو وسمها [I]/PROPOSED.

### ملاحظات غير مانعة
1. **التحذيرات في المصنع وحده:** الدالة العامة `create()` تتيح تجاوز المصنع دون توليد تحذيرات UNKNOWN، فعبارة "Sole producer" غير مفروضة برمجيًا (المصنع 11؛ ‏`ProjectionQueryConstraints.php` 43–86). لا يُعد هذا فقدًا صامتًا، لأن `UNKNOWN` يبقى ظاهرًا في `facet_constraints`.
2. **تكرار التحذير:** لا يُزال تكرار تحذير `QUERY_CONSTRAINT_UNKNOWN` إذا مرّره المستدعي مسبقًا (المصنع 71).
3. **الرموز الرقمية:** رموز facet الرقمية تتحول إلى مفاتيح مصفوفة صحيحة (المصنع 42–48؛ ‏`ProjectionQueryConstraints.php` 60–67). يعمل ذلك لأن الملفات لا تعلن `strict_types`، وإعلانه مستقبلًا سيكسر هذا المسار.
4. **فحص الموضوع بعد فرز البُعد:** السجلات من أبعاد أخرى تُتجاهل قبل فحص موضوعها (المصنع 89–91 مقابل 92)، فلا يُرفض سجل `content_about` لموضوع أجنبي.
5. **مصدر `retrieval_method`:** هو "المستدعي" في ADR (سطر 5657) والمصنع (سطر 31)، بينما الحزمة تقول "Path" (سطر 102). الانحراف موثّق في ADR، لكنه لم يُذكر صراحة كانحراف.
6. **فحص المحاذاة مرتين:** المصنع 34 و`ProjectionQueryConstraints.php` 51. لا ضرر.
